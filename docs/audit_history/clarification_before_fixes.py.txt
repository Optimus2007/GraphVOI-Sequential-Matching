"""Allocation-aware counterfactual asking with explicit, bounded approximations."""
import itertools
import json
import math

from kit import HARD, eligibility
from .answers import apply_answer
from .core import (ask_cost, available_members, first_asks, legal_asks, optimal_matching,
                   past_pairs, public_pairs, weighted_edges)


class MatchingContext:
    """Cache unaffected edges; recompute only edges touching hypothetical answers."""
    def __init__(self, state, scorer):
        self.state, self.scorer = state, scorer
        self.members = {m['member_id']: m for m in available_members(state)}
        self.past = past_pairs(state)
        self.edges = weighted_edges(state, scorer)
        self.value = optimal_matching(self.edges)[1]

    def updated(self, world):
        members = {m['member_id']: m for m in world['members']
                   if m['member_id'] in self.members}
        changed = {i for i in self.members if members[i]['fields'] != self.members[i]['fields']}
        edges = {p: v for p, v in self.edges.items() if not changed.intersection(p)}
        for ident in sorted(changed):
            for other in sorted(members):
                if ident == other:
                    continue
                pair = tuple(sorted((ident, other)))
                if pair in self.past or pair in edges:
                    continue
                a, b = (members[i] for i in pair)
                if eligibility(a, b)['status'] == 'feasible':
                    value = self.scorer(a, b, world['day'])
                    if value > 0:
                        edges[pair] = value
        return edges


def action_key(action):
    return tuple((a['member_id'], a['field']) for a in action)


def candidate_actions(state, scorer, paired=True, frontier=16):
    legal = legal_asks(state)
    potential = {}
    relevant = []
    for a, b, status in public_pairs(state, unresolved=True):
        if any(m['field_status'][k] == 'declined' for m in (a, b) for k in HARD):
            continue
        weight = scorer(a, b, state['day'])
        for m in (a, b):
            potential[m['member_id']] = potential.get(m['member_id'], 0) + weight
        relevant.append((a['member_id'], b['member_id'], weight, status))
    # Identical frontier for all selection strategies. No member IDs as features.
    ranked_singles = sorted(([a] for a in legal if potential.get(a['member_id'], 0) > 0),
                     key=lambda action: (-potential[action[0]['member_id']] / ask_cost(action[0]),
                                         action_key(action)))
    # Reserve half the shared frontier for hard completion. Otherwise seven
    # cheap soft fields on a blocked member can crowd out every hard bundle.
    hard = [a for a in ranked_singles if a[0]['field'] == 'constraints'][:frontier // 2]
    soft = [a for a in ranked_singles if a[0]['field'] != 'constraints'][:frontier - len(hard)]
    singles = hard + soft
    actions = singles[:]
    if paired:
        constraints = {a['member_id']: a for a in legal if a['field'] == 'constraints'}
        pairs = []
        for a, b, weight, status in relevant:
            if status == 'needs_clarification' and a in constraints and b in constraints:
                pairs.append((weight, [constraints[a], constraints[b]]))
        actions.extend(action for _, action in sorted(pairs, key=lambda x: (-x[0], action_key(x[1])))[:8])
    return actions


def counterfactuals(state, action, model):
    """Exact Cartesian expectation over each bounded empirical answer support."""
    worlds = [(1.0, state)]
    for ask in action:
        next_worlds = []
        for mass, world in worlds:
            for p, answer in model.outcomes(world, ask):
                next_worlds.append((mass * p, apply_answer(world, ask, answer)))
        worlds = next_worlds
    return worlds


def graphvoi_value(state, action, scorer, model, penalty=0.0, context=None):
    context = context or MatchingContext(state, scorer)
    baseline = context.value
    expected = math.fsum(p * optimal_matching(context.updated(world))[1]
                         for p, world in counterfactuals(state, action, model))
    return expected - baseline - penalty * sum(map(ask_cost, action))


def question_score(state, action, scorer, model, strategy, penalty=0.0, context=None):
    if strategy == 'graphvoi':
        return graphvoi_value(state, action, scorer, model, penalty, context)
    if strategy == 'degree_first':
        context = context or MatchingContext(state, scorer)
        before = len(context.edges)
        return math.fsum(p * (len(context.updated(world)) - before)
                         for p, world in counterfactuals(state, action, model))
    if strategy == 'uncertainty_first':
        # Sum conditional answer entropies over coordinates, weighted by
        # disclosure probability. Bundle costs are handled by shared VOI/unit
        # packing. This measures uncertainty, not merely missing-field count.
        score = 0.0
        for ask in action:
            outcomes = model.outcomes(state, ask)
            keys = sorted({k for _, answer in outcomes for k in answer})
            for key in keys:
                masses = {}
                for p, answer in outcomes:
                    if key in answer:
                        value = json.dumps(answer[key], sort_keys=True)
                        masses[value] = masses.get(value, 0) + p
                disclosed = sum(masses.values())
                if disclosed:
                    score += -sum(p * math.log(p / disclosed) for p in masses.values() if p > 0)
        return score
    raise ValueError('Unknown clarification strategy')


def select_asks(state, scorer, model, strategy='graphvoi', paired=True,
                frontier=16, penalty=0.0, diagnostics=None):
    """Static batch ranking by VOI/unit, duplicate-free packing within budget.

    Each action's matching counterfactual is true; a static greedy batch is an
    approximation to joint batch VOI and can contain redundant information.
    Complementary two-member actions explicitly cover zero-singleton gain cases.
    No wall-clock-dependent choices: fixed frontier/support yield deterministic work.
    """
    # A deterministic exceptional dense-graph guard protects validity. This is
    # explicitly a first-legal-hard-ask fallback, not a GraphVOI estimate. Apply
    # the same guard to all three controlled clarification methods.
    if len(weighted_edges(state, scorer)) > 600:
        if diagnostics is not None:
            diagnostics['fallback'] = 'dense_feasible_graph_over_600_edges'
        return first_asks(state)
    actions = candidate_actions(state, scorer, paired, frontier)
    context = MatchingContext(state, scorer)
    ranked = []
    for action in actions:
        cost = sum(map(ask_cost, action))
        if cost > state['ask_budget_remaining']:
            continue
        score = question_score(state, action, scorer, model, strategy, penalty, context)
        if score > 1e-12:
            ranked.append((-score / cost, action_key(action), action))
    chosen, seen, budget = [], set(), state['ask_budget_remaining']
    for _, key, action in sorted(ranked):
        cost = sum(map(ask_cost, action))
        if not seen.intersection(key) and cost <= budget:
            chosen.extend(action)
            seen.update(key)
            budget -= cost
    return chosen
