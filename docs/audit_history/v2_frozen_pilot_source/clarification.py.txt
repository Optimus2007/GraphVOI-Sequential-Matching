"""Allocation-aware counterfactual asking with explicit, bounded approximations."""
import itertools
import json
import math

from kit import HARD, eligibility
from .answers import apply_answer
from .core import (ask_cost, available_members, first_asks, legal_asks, optimal_matching,
                   past_pairs, public_pairs, weighted_edges)


class MatchingContext:
    """Cache unaffected edges; recompute only edges touching hypothetical answers."""
    def __init__(self, state, scorer):
        self.state, self.scorer = state, scorer
        self.members = {m['member_id']: m for m in available_members(state)}
        self.past = past_pairs(state)
        self.edges = weighted_edges(state, scorer)
        self.value = optimal_matching(self.edges)[1]

    def updated(self, world):
        members = {m['member_id']: m for m in world['members']
                   if m['member_id'] in self.members}
        changed = {i for i in self.members if members[i]['fields'] != self.members[i]['fields']}
        edges = {p: v for p, v in self.edges.items() if not changed.intersection(p)}
        for ident in sorted(changed):
            for other in sorted(members):
                if ident == other:
                    continue
                pair = tuple(sorted((ident, other)))
                if pair in self.past or pair in edges:
                    continue
                a, b = (members[i] for i in pair)
                if eligibility(a, b)['status'] == 'feasible':
                    value = self.scorer(a, b, world['day'])
                    if value > 0:
                        edges[pair] = value
        return edges


def action_key(action):
    return tuple((a['member_id'], a['field']) for a in action)


def candidate_actions(state, scorer, paired=True, frontier=16):
    legal = legal_asks(state)
    potential = {}
    relevant = []
    for a, b, status in public_pairs(state, unresolved=True):
        if any(m['field_status'][k] == 'declined' for m in (a, b) for k in HARD):
            continue
        weight = scorer(a, b, state['day'])
        for m in (a, b):
            potential[m['member_id']] = potential.get(m['member_id'], 0) + weight
        relevant.append((a['member_id'], b['member_id'], weight, status))
    # Identical frontier for all selection strategies. No member IDs as features.
    ranked_singles = sorted(([a] for a in legal if potential.get(a['member_id'], 0) > 0),
                     key=lambda action: (-potential[action[0]['member_id']] / ask_cost(action[0]),
                                         action_key(action)))
    # Reserve half the shared frontier for hard completion. Otherwise seven
    # cheap soft fields on a blocked member can crowd out every hard bundle.
    hard = [a for a in ranked_singles if a[0]['field'] == 'constraints'][:frontier // 2]
    soft = [a for a in ranked_singles if a[0]['field'] != 'constraints'][:frontier - len(hard)]
    singles = hard + soft
    actions = singles[:]
    if paired:
        constraints = {a['member_id']: a for a in legal if a['field'] == 'constraints'}
        pairs = []
        for a, b, weight, status in relevant:
            if status == 'needs_clarification' and a in constraints and b in constraints:
                pairs.append((weight, [constraints[a], constraints[b]]))
        # Reserve four paired slots for direct reciprocal hard completion, then
        # consider joint questions across the singleton frontier. Information
        # can improve an alternating allocation without the asked members being
        # partners, and two soft fields can also be complementary.
        direct=[action for _,action in sorted(pairs,key=lambda x:(-x[0],action_key(x[1])))[:4]]
        general=[]
        included={action_key(action) for action in direct}
        for left,right in itertools.combinations(singles,2):
            action=sorted(left+right,key=lambda q:(q['member_id'],q['field']))
            key=action_key(action)
            if key in included:continue
            ids={q['member_id'] for q in action}
            priority=sum(potential[i] for i in ids)/sum(map(ask_cost,action))
            general.append((-priority,key,action))
        actions.extend(direct+[action for _,_,action in sorted(general)[:8-len(direct)]])
    return actions


def counterfactuals(state, action, model):
    """Exact Cartesian expectation over each bounded empirical answer support."""
    worlds = [(1.0, state)]
    for ask in action:
        next_worlds = []
        for mass, world in worlds:
            for p, answer in model.outcomes(world, ask):
                next_worlds.append((mass * p, apply_answer(world, ask, answer)))
        worlds = next_worlds
    return worlds


def graphvoi_value(state, action, scorer, model, penalty=0.0, context=None):
    context = context or MatchingContext(state, scorer)
    baseline = context.value
    expected = math.fsum(p * optimal_matching(context.updated(world))[1]
                         for p, world in counterfactuals(state, action, model))
    return expected - baseline - penalty * sum(map(ask_cost, action))


def question_score(state, action, scorer, model, strategy, penalty=0.0, context=None):
    if strategy == 'graphvoi':
        return graphvoi_value(state, action, scorer, model, penalty, context)
    if strategy == 'degree_first':
        context = context or MatchingContext(state, scorer)
        before = len(context.edges)
        return math.fsum(p * (len(context.updated(world)) - before)
                         for p, world in counterfactuals(state, action, model))
    if strategy == 'uncertainty_first':
        # Sum conditional answer entropies over coordinates, weighted by
        # disclosure probability. Bundle costs are handled by shared VOI/unit
        # packing. This measures uncertainty, not merely missing-field count.
        score = 0.0
        for ask in action:
            outcomes = model.outcomes(state, ask)
            keys = sorted({k for _, answer in outcomes for k in answer})
            for key in keys:
                masses = {}
                for p, answer in outcomes:
                    if key in answer:
                        value = json.dumps(answer[key], sort_keys=True)
                        masses[value] = masses.get(value, 0) + p
                disclosed = sum(masses.values())
                if disclosed:
                    score += -sum(p * math.log(p / disclosed) for p in masses.values() if p > 0)
        return score
    raise ValueError('Unknown clarification strategy')


def joint_value_reference(state, asks, scorer, model, penalty=0.0):
    """Unbounded exact finite-support reference; unique asks, charge each once."""
    unique = { (q['member_id'],q['field']): q for q in asks }
    return graphvoi_value(state, [unique[k] for k in sorted(unique)], scorer, model, penalty)


def merge_worlds(worlds, fingerprint=None):
    """Coalesce identical public counterfactual records without discarding mass."""
    merged={}
    for mass,world in worlds:
        key=(fingerprint(world) if fingerprint else
             tuple((m['member_id'],json.dumps(m['fields'],sort_keys=True),
                    json.dumps(m['field_status'],sort_keys=True)) for m in world['members']))
        if key not in merged:merged[key]=[mass,world]
        else:merged[key][0]+=mass
    return [(p,w) for p,w in merged.values()]


def compress_worlds(worlds, limit):
    """Deterministic systematic probability quadrature, exact below support cap.

    It compresses a joint distribution, never subtracts independent VOI scores.
    Order is deterministic. Support truncation may bias expectations; no global
    optimality guarantee is claimed. The exact reference does not use this cap.
    """
    if len(worlds) <= limit:
        return worlds
    selected=[]; index=0; cumulative=worlds[0][0]
    for j in range(limit):
        target=(j+.5)/limit
        while target>cumulative and index<len(worlds)-1:
            index+=1; cumulative+=worlds[index][0]
        selected.append((1/limit,worlds[index][1]))
    return selected


def select_asks(state, scorer, model, strategy='graphvoi', paired=True,
                frontier=16, penalty=0.0, diagnostics=None, world_limit=8,
                evaluation_limit=256):
    """Greedy conditional marginal acquisition over joint answer worlds.

    Whole paired candidates preserve complementarity. Overlapping asks become
    residual actions, never duplicate cost. Eight worlds and 256 outcome evaluations
    bound production work identically for the three strategies. Exact examples can
    override these limits. Exhausting work stops safely with the already chosen
    legal asks; this is not a fabricated VOI or a wall-clock-dependent fallback.
    """
    if strategy not in ('graphvoi','degree_first','uncertainty_first'):
        raise ValueError('Unknown clarification strategy')
    if world_limit<1 or evaluation_limit<1:
        raise ValueError('Positive world and work limits required')
    if len(weighted_edges(state, scorer)) > 600:
        if diagnostics is not None:
            diagnostics['fallback'] = 'dense_feasible_graph_over_600_edges'
        return first_asks(state)
    actions=candidate_actions(state,scorer,paired,frontier)
    root=MatchingContext(state,scorer)
    # Worlds are immutable within acquisition: apply_answer copies a changed
    # member. Retain the object alongside its full-record fingerprint to prevent
    # id reuse. This memo is invocation-local and includes every public member
    # attribute; it is an acceleration of serialization, not an answer belief.
    record_keys={}
    def fingerprint(world):
        keys=[]
        for m in world['members']:
            ident=id(m)
            if ident not in record_keys:
                record_keys[ident]=(m,json.dumps(m,sort_keys=True))
            keys.append(record_keys[ident][1])
        return (world['day'],tuple(keys))
    # Current joint-world values cached within this invocation only.
    value_cache={}
    def value(world):
        key=fingerprint(world)
        if key not in value_cache:
            edges=root.updated(world)
            value_cache[key]=(optimal_matching(edges)[1],len(edges))
        return value_cache[key]
    worlds=[(1.0,state)]; chosen=[]; seen=set(); budget=state['ask_budget_remaining']; work=0
    while budget>0:
        residuals={}
        for action in actions:
            rest=[q for q in action if (q['member_id'],q['field']) not in seen]
            key=action_key(rest)
            if rest and sum(map(ask_cost,rest))<=budget:residuals[key]=rest
        if not residuals:break
        best=None
        baseline=math.fsum(p*value(w)[0 if strategy=='graphvoi' else 1] for p,w in worlds) if strategy!='uncertainty_first' else 0
        # Canonical enumeration prevents a work cap from depending on wall time.
        for key,action in sorted(residuals.items()):
            expanded=[]
            for mass,world in worlds:
                expanded.extend((mass*p,w) for p,w in counterfactuals(world,action,model))
            expanded=merge_worlds(expanded,fingerprint)
            if work+len(expanded)>evaluation_limit:
                continue
            work+=len(expanded)
            cost=sum(map(ask_cost,action))
            if strategy=='uncertainty_first':
                gain=math.fsum(p*question_score(w,action,scorer,model,strategy) for p,w in worlds)
            else:
                column=0 if strategy=='graphvoi' else 1
                gain=math.fsum(p*value(w)[column] for p,w in expanded)-baseline
                if strategy=='graphvoi':gain-=penalty*cost
            rank=(-gain/cost,key)
            if gain>1e-12 and (best is None or rank<best[0]):best=(rank,action,expanded)
        if best is None:break
        _,action,expanded=best
        chosen.extend(action);seen.update(action_key(action));budget-=sum(map(ask_cost,action))
        worlds=compress_worlds(expanded,world_limit)
    if diagnostics is not None:
        diagnostics.update(outcome_evaluations=work,joint_worlds=len(worlds),
                           evaluation_limit=evaluation_limit,world_limit=world_limit)
    return chosen
